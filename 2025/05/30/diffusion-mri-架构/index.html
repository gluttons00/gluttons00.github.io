CrossAttention

class CrossAttention(torch.nn.Module):
def init(self, dim_q, dim_kv):
super().init()
self.dim_q = dim_q # 查询的维度
self.q = torch.nn.Linear(dim_q, dim_q, bias=False) # 查询线性变换
self.k = torch.nn.Linear(dim_kv, dim_q, bias=False) # 键线性变换
self.v = torch.nn.Linear(dim_kv, dim_q, bias=False) # 值线性变换
self.out = torch.nn.Linear(dim_q, dim_q) # 输出线性变换
